create table public.orders (
id bigint generated by default as identity,
user_id bigint not null,
status text not null default 'pending',
total_cents integer not null default 0,
currency char(3) not null default 'EUR',
created_at timestamptz not null default now(),
shipped_at timestamptz,
coupon_code text,
channel text not null default 'web',
country char(2),
metadata jsonb default '{}',
note text,
refunded_cents integer not null default 0,
updated_at timestamptz not null default now(),
constraint orders_pkey primary key (id),
constraint orders_user_fk foreign key (user_id)
references public.users (id) on delete restrict,
constraint orders_status_chk check (status in
('pending','paid','shipped','refunded','cancelled'))
);
create index orders_created_idx
on public.orders using btree (created_at desc);
